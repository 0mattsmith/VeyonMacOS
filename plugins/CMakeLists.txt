file(GLOB plugins RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/*)

option(WITH_LDAP "Build LDAP plugin" ON)

foreach(plugin ${plugins})
	if(IS_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/${plugin})
		if(plugin STREQUAL "ldap")
			if(NOT WITH_LDAP)
				continue()
			endif()
			if(NOT VEYON_BUILD_LINUX AND NOT VEYON_BUILD_WINDOWS)
				# LDAP development files are optional on other platforms (e.g. macOS)
				find_package(Ldap QUIET)
				find_package(Sasl2 QUIET)
				if(NOT Ldap_FOUND OR NOT Sasl2_FOUND)
					message(WARNING "LDAP or SASL2 development files not found - not building LDAP plugin")
					continue()
				endif()
			endif()
		endif()
		add_subdirectory(${plugin})
	endif()
endforeach()
